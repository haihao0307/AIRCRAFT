name: Aircraft native R01 HTML review
on:
  push:
    branches: [feature/b24-weapons-mother-v1]
    paths: [native/weapons-visual/web/r01/**, .github/workflows/wm-native-web-r01.yml]
  workflow_dispatch:
permissions:
  contents: read
jobs:
  review:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@v4
        with:
          sparse-checkout: native/weapons-visual/web/r01
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - name: Fixed rendering library
        shell: bash
        run: |
          set -euo pipefail
          mkdir -p vendor dist
          base=https://raw.githubusercontent.com/mrdoob/three.js/r170
          curl -fLsS --retry 2 "$base/build/three.module.js" -o vendor/three.module.js
          curl -fLsS --retry 2 "$base/examples/jsm/controls/OrbitControls.js" -o vendor/OrbitControls.js
          curl -fLsS --retry 2 "$base/LICENSE" -o vendor/THREE_LICENSE.txt
          python native/weapons-visual/web/r01/build.py --vendor vendor --output dist/AIRCRAFT_NATIVE_R01.html
          cp -r native/weapons-visual/web/r01 dist/source
          cp -r vendor dist/vendor
          git rev-parse HEAD > dist/SOURCE_COMMIT.txt
          node --check native/weapons-visual/web/r01/model.js
          node --check native/weapons-visual/web/r01/surface.js
          node --check native/weapons-visual/web/r01/app.js
      - name: Install browser
        run: |
          python -m pip install playwright==1.55.0 Pillow==11.3.0
          python -m playwright install --with-deps chromium
          sudo apt-get update -qq && sudo apt-get install -y fonts-noto-cjk
      - name: Real scene and user controls
        run: python native/weapons-visual/web/r01/qa.py dist/AIRCRAFT_NATIVE_R01.html --out dist/qa
      - uses: actions/upload-artifact@v4
        if: always()
        with:
          name: WM_NATIVE_WEB_R01_${{ github.sha }}
          path: dist/
          retention-days: 30
          if-no-files-found: error
